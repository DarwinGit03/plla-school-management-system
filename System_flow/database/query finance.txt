CREATE TABLE fee_tuition (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,

    configuration_id INT UNSIGNED NOT NULL,

    payment_sequence INT UNSIGNED NOT NULL,

    payment_label VARCHAR(100) NOT NULL,

    payment_date DATE NULL,

    tuition_fee DECIMAL(12,2) NOT NULL DEFAULT 0.00,

    status ENUM(
        'active',
        'inactive'
    ) NOT NULL DEFAULT 'active',

    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,

    updated_at DATETIME NULL
        DEFAULT NULL
        ON UPDATE CURRENT_TIMESTAMP,

    UNIQUE KEY uq_payment_schedule (
        configuration_id,
        payment_sequence
    ),

    CONSTRAINT fk_fee_tuition_configuration
        FOREIGN KEY (configuration_id)
        REFERENCES fee_configurations(id)
        ON DELETE CASCADE
        ON UPDATE CASCADE,

    INDEX idx_fee_tuition_configuration (
        configuration_id
    )

) ENGINE=InnoDB;
